# ============================================================
# PROPOSED FINAL MODEL
# Stateful Real-Time Unified Vehicle Ranking
#
# ============================================================
# FINAL FIXED PARAMETERS
# ------------------------------------------------------------
# Minimum Arrival Slack = 25 min
# Maximum Work Gap      = 24 h
#
# ============================================================
# DEFINITIONS
# ------------------------------------------------------------
#
# Work Gap
#   = Order Time - Vehicle Available Time
#
# Candidate Work-Gap Gate
#   0 < Work Gap <= 24 h
#
# Empty Travel
#   = Current Vehicle Location -> New Order Pickup
#
# Estimated Arrival
#   = Vehicle Available Time + Empty Travel Time
#
# Arrival Slack
#   = Order Time - Estimated Arrival
#
# Arrival Gate
#   Arrival Slack >= 25 min
#
# Ranking
#   1. Minimum Empty Distance
#   2. Minimum Empty Travel Time
#   3. Maximum Arrival Slack
#   4. Vehicle ID
#
# Stateful Update
#   Recommended vehicle:
#       location       -> order dropoff
#       available_time -> estimated order finish time
#
# ============================================================
# IMPORTANT
# ------------------------------------------------------------
# - Future FIXED Schedule 사용하지 않음
# - Next FIXED Return Constraint 사용하지 않음
# - Weighted Objective 사용하지 않음
# - Fallback Cost 사용하지 않음
# - MILP 아님
#
# HISTORY_PATH의 파일명에 FIXED가 들어가지만
# 현재 코드는 미래 FIXED 스케줄을 활용하지 않고,
# 해당 파일 안의 과거 차량 운행 이력만
# State 복원용 History Source로 사용함.
# ============================================================

def clean_text(series):

    s = (
        series
        .astype("string")
        .str.strip()
    )

    invalid = {
        "",
        "nan",
        "NaN",
        "None",
        "<NA>",
        "NULL",
        "null",
    }

    return s.mask(
        s.isin(invalid)
    )


def valid_text(x):

    if pd.isna(x):
        return False

    s = str(x).strip()

    if not s:
        return False

    if s.lower() in {
        "nan",
        "none",
        "<na>",
        "null",
    }:
        return False

    return True


def select_sheet(
    path,
    candidates,
):

    names = (
        pd.ExcelFile(path)
        .sheet_names
    )

    for name in candidates:

        if name in names:
            return name

    raise KeyError(
        f"\n{path.name}에서 시트를 찾지 못했습니다."
        f"\n후보: {candidates}"
        f"\n실제 시트: {names}"
    )


def select_col(
    df,
    candidates,
    required=True,
):

    for col in candidates:

        if col in df.columns:
            return col

    if required:

        raise KeyError(
            "\n필요 컬럼을 찾지 못했습니다."
            f"\n후보: {candidates}"
            f"\n실제 컬럼:\n{list(df.columns)}"
        )

    return None


# ============================================================
# 4. WEIGHT GATE FUNCTION
# ============================================================

def judge_weight_gate(
    container_type,
    weight_kg,
):

    if not valid_text(container_type):
        return "UNKNOWN_TYPE"

    ctype = (
        str(container_type)
        .strip()
        .upper()
    )

    if ctype not in WEIGHT_GATE:
        return "UNKNOWN_TYPE"

    if not np.isfinite(weight_kg):
        return "UNKNOWN_WEIGHT"

    rule = WEIGHT_GATE[
        ctype
    ]

    warn = rule[
        "warn_kg"
    ]

    block = rule[
        "block_kg"
    ]

    if weight_kg >= block:
        return "BLOCK"

    if warn is None:
        return "OK"

    if weight_kg >= warn:
        return "WARN"

    return "OK"

                obs_ptr[
                    v_idx
                ]
            )

            next_p = (
                p + 1
            )


            while (
                next_p
                <
                len(times)
                and
                times[
                    next_p
                ]
                <
                order_time_np
            ):

                p = (
                    next_p
                )

                next_p += 1


            obs_ptr[
                v_idx
            ] = p


        # ====================================================
        # B. WEIGHT BLOCK
        # ====================================================

        if gate_status == "BLOCK":

            rows.append({

                "order_id":
                    ORDER_ID[i],

                "order_time":
                    pd.Timestamp(
                        order_time_np
                    ),

                "minimum_arrival_slack_min":
                    MIN_ARRIVAL_SLACK_MIN,

                "max_work_gap_h":
                    MAX_WORK_GAP_H,

                "recommendation_status":
                    "BLOCK",

                "recommended_vehicle":
                    pd.NA,

                "candidate_count":
                    0,

                "container_type":
                    ORDER_CONTAINER[i],

                "weight_kg":
                    ORDER_WEIGHT_KG[i],

                "weight_gate":
                    gate_status,

                "actual_vehicle":
                    ORDER_ACTUAL_VEHICLE[i],

                "actual_dispatch":
                    ORDER_ACTUAL_DISPATCH[i],

                "actual_empty_distance_km":
                    ORDER_ACTUAL_EMPTY_DIST[i],
            })

            continue


        # ====================================================
        # C. CANDIDATE SEARCH
        # ====================================================

        best_idx = (
            -1
        )

        best_distance = (
            np.inf
        )

        best_empty_time_h = (
            np.inf
        )

        best_slack_min = (
            -np.inf
        )

        best_work_gap_h = (
            np.nan
        )

        best_location = (
            None
        )

        best_available_ns = (
            None
        )

        best_arrival_ns = (
            None
        )

        best_source = (
            None
        )

        best_state_age_h = (
            np.nan
        )


        candidate_count = (
            0
        )

        excluded_by_work_gap = (
            0
        )

        excluded_by_slack = (
            0
        )


        # ====================================================
        # VEHICLE LOOP
        # ====================================================

        for v_idx, vehicle in enumerate(
            VEHICLES
        ):

            # ------------------------------------------------
            # C-1. LATEST OBSERVED STATE
            # ------------------------------------------------

            p = (
                obs_ptr[
                    v_idx
                ]
            )


            if p >= 0:

                obs_time_np = (
                    OBS_TIME[
                        v_idx
                    ][p]
                )

                obs_time_ns = (
                    obs_time_np
                    .astype(
                        "datetime64[ns]"
                    )
                    .astype(
                        np.int64
                    )
                )

                obs_location = (
                    OBS_LOCATION[
                        v_idx
                    ][p]
                )

                has_observed = (
                    True
                )


            else:

                obs_time_ns = (
                    INT_MIN
                )

                obs_location = (
                    None
                )

                has_observed = (
                    False
                )


            # ------------------------------------------------
            # C-2. OBSERVED VS SIMULATED
            # ------------------------------------------------

            if (
                sim_has_state[
                    v_idx
                ]
                and
                (
                    not has_observed
                    or
                    sim_available_ns[
                        v_idx
                    ]
                    >=
                    obs_time_ns
                )
            ):

                available_ns = (
                    sim_available_ns[
                        v_idx
                    ]
                )

                location = (
                    sim_location[
                        v_idx
                    ]
                )

                source = (
                    "simulated_target"
                )

                state_age_h = (
                    np.nan
                )


            elif has_observed:

                available_ns = (
                    obs_time_ns
                )

                location = (
                    obs_location
                )

                source = (
                    "observed_history"
                )

                state_age_h = (
                    order_time_ns
                    -
                    available_ns
                ) / ONE_HOUR_NS


            else:

                continue


            # ------------------------------------------------
            # C-3. OBSERVED STATE FRESHNESS
            # ------------------------------------------------

            if (
                source
                ==
                "observed_history"
            ):

                state_age_ns = (
                    order_time_ns
                    -
                    available_ns
                )


                if (
                    state_age_ns
                    >
                    STATE_STALE_CUTOFF_NS
                ):

                    continue


            # =================================================
            # C-4. WORK GAP
            #
            # Work Gap
            #   = Order Time - Vehicle Available Time
            # =================================================

            work_gap_ns = (
                order_time_ns
                -
                available_ns
            )


            # ------------------------------------------------
            # Work Gap <= 0
            # ------------------------------------------------

            if (
                work_gap_ns
                <= 0
            ):

                excluded_by_work_gap += 1

                continue


            # ------------------------------------------------
            # Work Gap > 24 h
            # ------------------------------------------------

            if (
                work_gap_ns
                >
                MAX_WORK_GAP_NS
            ):

                excluded_by_work_gap += 1

                continue


            work_gap_h = (
                work_gap_ns
                /
                ONE_HOUR_NS
            )


            # =================================================
            # C-5. CURRENT LOCATION -> PICKUP OD
            # =================================================

            if (
                location
                ==
                pickup
            ):

                empty_distance = (
                    0.0
                )

                empty_time_h = (
                    0.0
                )


            else:

                od_value = (
                    OD_MAP.get(
                        (
                            location,
                            pickup,
                        )
                    )
                )


                if od_value is None:

                    continue


                (
                    empty_distance,
                    empty_time_h,
                ) = (
                    od_value
                )


            if (
                not np.isfinite(
                    empty_distance
                )
                or
                not np.isfinite(
                    empty_time_h
                )
            ):

                continue


            # =================================================
            # C-6. ESTIMATED ARRIVAL
            #
            # Available Time + Empty Travel Time
            # =================================================

            travel_ns = int(
                empty_time_h
                *
                ONE_HOUR_NS
            )


            estimated_arrival_ns = (
                available_ns
                +
                travel_ns
            )


            # =================================================
            # C-7. ARRIVAL SLACK
            #
            # Order Time - Estimated Arrival
            # =================================================

            arrival_slack_ns = (
                order_time_ns
                -
                estimated_arrival_ns
            )


            arrival_slack_min = (
                arrival_slack_ns
                /
                ONE_MIN_NS
            )


            # =================================================
            # C-8. ARRIVAL SLACK >= 25 MIN
            # =================================================

            if (
                arrival_slack_ns
                <
                MIN_SLACK_NS
            ):

                excluded_by_slack += 1

                continue


            candidate_count += 1


            # =================================================
            # C-9. RANKING
            #
            # 1. Minimum Empty Distance
            # 2. Minimum Empty Time
            # 3. Maximum Arrival Slack
            # 4. Vehicle ID
            # =================================================

            is_better = (
                False
            )


            if (
                empty_distance
                <
                best_distance
            ):

                is_better = (
                    True
                )


            elif (
                empty_distance
                ==
                best_distance
            ):

                if (
                    empty_time_h
                    <
                    best_empty_time_h
                ):

                    is_better = (
                        True
                    )


                elif (
                    empty_time_h
                    ==
                    best_empty_time_h
                ):

                    if (
                        arrival_slack_min
                        >
                        best_slack_min
                    ):

                        is_better = (
                            True
                        )


                    elif (
                        arrival_slack_min
                        ==
                        best_slack_min
                        and
                        best_idx >= 0
                        and
                        vehicle
                        <
                        VEHICLES[
                            best_idx
                        ]
                    ):

                        is_better = (
                            True
                        )


            if is_better:

                best_idx = (
                    v_idx
                )

                best_distance = float(
                    empty_distance
                )

                best_empty_time_h = float(
                    empty_time_h
                )

                best_slack_min = float(
                    arrival_slack_min
                )

                best_work_gap_h = float(
                    work_gap_h
                )

                best_location = (
                    location
                )

                best_available_ns = int(
                    available_ns
                )

                best_arrival_ns = int(
                    estimated_arrival_ns
                )

                best_source = (
                    source
                )

                best_state_age_h = (
                    state_age_h
                )


        # ====================================================
        # D. NO FEASIBLE
        # ====================================================

        if best_idx < 0:

            rows.append({

                "order_id":
                    ORDER_ID[i],

                "order_time":
                    pd.Timestamp(
                        order_time_np
                    ),

                "minimum_arrival_slack_min":
                    MIN_ARRIVAL_SLACK_MIN,

                "max_work_gap_h":
                    MAX_WORK_GAP_H,

                "recommendation_status":
                    "NO_FEASIBLE",

                "recommended_vehicle":
                    pd.NA,

                "candidate_count":
                    0,

                "excluded_by_work_gap_count":
                    excluded_by_work_gap,

                "excluded_by_slack_count":
                    excluded_by_slack,

                "container_type":
                    ORDER_CONTAINER[i],

                "weight_kg":
                    ORDER_WEIGHT_KG[i],

                "weight_gate":
                    gate_status,

                "actual_vehicle":
                    ORDER_ACTUAL_VEHICLE[i],

                "actual_dispatch":
                    ORDER_ACTUAL_DISPATCH[i],

                "actual_empty_distance_km":
                    ORDER_ACTUAL_EMPTY_DIST[i],
            })


        # ====================================================
        # E. RECOMMENDED
        # ====================================================

        else:

            best_vehicle = (
                VEHICLES[
                    best_idx
                ]
            )


            # ------------------------------------------------
            # STATEFUL UPDATE
            # ------------------------------------------------

            sim_has_state[
                best_idx
            ] = (
                True
            )


            sim_location[
                best_idx
            ] = (
                dropoff
            )


            sim_available_ns[
                best_idx
            ] = (
                finish_ns
            )


            # ------------------------------------------------
            # ACTUAL COMPARISON
            # ------------------------------------------------

            actual_empty = (
                ORDER_ACTUAL_EMPTY_DIST[
                    i
                ]
            )


            if np.isfinite(
                actual_empty
            ):

                saving_distance = (
                    actual_empty
                    -
                    best_distance
                )


                if actual_empty > 0:

                    order_saving_rate = (
                        saving_distance
                        /
                        actual_empty
                    )

                else:

                    order_saving_rate = (
                        np.nan
                    )


            else:

                saving_distance = (
                    np.nan
                )

                order_saving_rate = (
                    np.nan
                )


            rows.append({

                # --------------------------------------------
                # SETTINGS
                # --------------------------------------------

                "minimum_arrival_slack_min":
                    MIN_ARRIVAL_SLACK_MIN,

                "max_work_gap_h":
                    MAX_WORK_GAP_H,

                # --------------------------------------------
                # ORDER
                # --------------------------------------------

                "order_id":
                    ORDER_ID[i],

                "order_time":
                    pd.Timestamp(
                        order_time_np
                    ),

                "order_pickup_rep":
                    pickup,

                "order_dropoff_rep":
                    dropoff,

                # --------------------------------------------
                # RECOMMENDATION
                # --------------------------------------------

                "recommendation_status":
                    "RECOMMENDED",

                "recommended_vehicle":
                    best_vehicle,

                "candidate_count":
                    candidate_count,

                "excluded_by_work_gap_count":
                    excluded_by_work_gap,

                "excluded_by_slack_count":
                    excluded_by_slack,

                # --------------------------------------------
                # VEHICLE STATE
                # --------------------------------------------

                "vehicle_before_location":
                    best_location,

                "vehicle_available_time":
                    pd.to_datetime(
                        best_available_ns
                    ),

                "state_source":
                    best_source,

                "observed_state_age_h":
                    best_state_age_h,

                # --------------------------------------------
                # WORK GAP
                # --------------------------------------------

                "work_gap_h":
                    best_work_gap_h,

                "work_gap_min":
                    best_work_gap_h
                    * 60.0,

                # --------------------------------------------
                # EMPTY TRAVEL
                # --------------------------------------------

                "recommended_empty_distance_km":
                    best_distance,

                "recommended_empty_time_h":
                    best_empty_time_h,

                "recommended_empty_time_min":
                    best_empty_time_h
                    * 60.0,

                # --------------------------------------------
                # ARRIVAL
                # --------------------------------------------

                "estimated_arrival_time":
                    pd.to_datetime(
                        best_arrival_ns
                    ),

                "arrival_slack_min":
                    best_slack_min,

                "arrival_slack_h":
                    best_slack_min
                    / 60.0,

                # --------------------------------------------
                # FINISH
                # --------------------------------------------

                "estimated_finish_time":
                    pd.Timestamp(
                        ORDER_FINISH[i]
                    ),

                # --------------------------------------------
                # WEIGHT
                # --------------------------------------------

                "container_type":
                    ORDER_CONTAINER[i],

                "weight_kg":
                    ORDER_WEIGHT_KG[i],

                "weight_gate":
                    gate_status,

                # --------------------------------------------
                # ACTUAL
                # --------------------------------------------

                "actual_vehicle":
                    ORDER_ACTUAL_VEHICLE[i],

                "actual_dispatch":
                    ORDER_ACTUAL_DISPATCH[i],

                "actual_empty_distance_km":
                    actual_empty,

                # --------------------------------------------
                # SAVING
                # --------------------------------------------

                "saving_distance_km":
                    saving_distance,

                "order_empty_saving_rate":
                    order_saving_rate,
            })


        # ====================================================
        # PROGRESS
        # ====================================================

        if (
            (i + 1)
            %
            PROGRESS_EVERY
            == 0
            or
            i + 1
            ==
            N_ORDERS
        ):

            elapsed = (
                perf_counter()
                -
                start
            )


            print(
                f"  {i + 1:>5,}"
                f" / {N_ORDERS:,}"
                f" 완료"
                f" | {elapsed:,.1f} sec"
            )


    result_df = (
        pd.DataFrame(
            rows
        )
    )


    elapsed = (
        perf_counter()
        -
        start
    )


    print(
        f"\nSimulation 완료"
        f" | {elapsed:,.1f} sec"
    )


    return result_df

# 17. VEHICLE-LEVEL SAVING TABLE
# ============================================================

def make_vehicle_saving_table(
    result_df,
):

    compare_df = result_df[
        (
            result_df[
                "recommendation_status"
            ]
            ==
            "RECOMMENDED"
        )
        &
        (
            result_df[
                "recommended_vehicle"
            ].notna()
        )
        &
        (
            result_df[
                "actual_empty_distance_km"
            ].notna()
        )
        &
        (
            result_df[
                "recommended_empty_distance_km"
            ].notna()
        )
    ].copy()


    if len(
        compare_df
    ) == 0:

        return (
            pd.DataFrame()
        )


    vehicle_df = (
        compare_df
        .groupby(
            "recommended_vehicle",
            dropna=False,
        )
        .agg(

            추천건수=(
                "order_id",
                "count"
            ),

            실제공차거리합_km=(
                "actual_empty_distance_km",
                "sum"
            ),

            추천공차거리합_km=(
                "recommended_empty_distance_km",
                "sum"
            ),

            실제공차거리평균_km=(
                "actual_empty_distance_km",
                "mean"
            ),

            추천공차거리평균_km=(
                "recommended_empty_distance_km",
                "mean"
            ),

        )
        .reset_index()
    )


    # --------------------------------------------------------
    # SAVING DISTANCE
    # --------------------------------------------------------

    vehicle_df[
        "공차절감거리_km"
    ] = (
        vehicle_df[
            "실제공차거리합_km"
        ]
        -
        vehicle_df[
            "추천공차거리합_km"
        ]
    )


    # --------------------------------------------------------
    # VEHICLE SAVING RATE
    # --------------------------------------------------------

    vehicle_df[
        "차량별공차절감률"
    ] = np.where(

        vehicle_df[
            "실제공차거리합_km"
        ]
        > 0,

        (
            vehicle_df[
                "공차절감거리_km"
            ]
            /
            vehicle_df[
                "실제공차거리합_km"
            ]
        ),

        np.nan,
    )


    vehicle_df[
        "차량별공차절감률_pct"
    ] = (
        vehicle_df[
            "차량별공차절감률"
        ]
        * 100.0
    )


    vehicle_df[
        "실제대비개선차량여부"
    ] = np.where(

        vehicle_df[
            "추천공차거리합_km"
        ]
        <
        vehicle_df[
            "실제공차거리합_km"
        ],

        "O",

        "X",
    )


    vehicle_df = (
        vehicle_df
        .sort_values(
            [
                "차량별공차절감률",
                "추천건수",
            ],
            ascending=[
                False,
                False,
            ],
            na_position="last",
        )
        .reset_index(
            drop=True
        )
    )


    return vehicle_df

# 18. EVALUATION
# ============================================================

def evaluate(
    result_df,
):

    total_n = (
        len(
            result_df
        )
    )


    blocked = (
        result_df[
            result_df[
                "recommendation_status"
            ]
            ==
            "BLOCK"
        ]
    )


    recommended = (
        result_df[
            result_df[
                "recommendation_status"
            ]
            ==
            "RECOMMENDED"
        ]
        .copy()
    )


    no_feasible = (
        result_df[
            result_df[
                "recommendation_status"
            ]
            ==
            "NO_FEASIBLE"
        ]
    )


    blocked_n = (
        len(
            blocked
        )
    )

    eligible_n = (
        total_n
        -
        blocked_n
    )

    rec_n = (
        len(
            recommended
        )
    )


    recommendation_rate = (
        rec_n
        /
        eligible_n

        if eligible_n > 0

        else np.nan
    )


    # ========================================================
    # RECOMMENDATION METRICS
    # ========================================================

    if rec_n > 0:

        avg_candidate = (
            recommended[
                "candidate_count"
            ].mean()
        )

        median_candidate = (
            recommended[
                "candidate_count"
            ].median()
        )


        avg_work_gap = (
            recommended[
                "work_gap_h"
            ].mean()
        )

        median_work_gap = (
            recommended[
                "work_gap_h"
            ].median()
        )

        q95_work_gap = (
            recommended[
                "work_gap_h"
            ].quantile(
                0.95
            )
        )


        avg_distance = (
            recommended[
                "recommended_empty_distance_km"
            ].mean()
        )

        median_distance = (
            recommended[
                "recommended_empty_distance_km"
            ].median()
        )


        zero_distance_rate = (
            recommended[
                "recommended_empty_distance_km"
            ]
            <=
            1e-9
        ).mean()


        avg_empty_time = (
            recommended[
                "recommended_empty_time_min"
            ].mean()
        )

        median_empty_time = (
            recommended[
                "recommended_empty_time_min"
            ].median()
        )


        avg_slack = (
            recommended[
                "arrival_slack_min"
            ].mean()
        )

        median_slack = (
            recommended[
                "arrival_slack_min"
            ].median()
        )

        q25_slack = (
            recommended[
                "arrival_slack_min"
            ].quantile(
                0.25
            )
        )

        q75_slack = (
            recommended[
                "arrival_slack_min"
            ].quantile(
                0.75
            )
        )


    else:

        avg_candidate = (
            np.nan
        )

        median_candidate = (
            np.nan
        )

        avg_work_gap = (
            np.nan
        )

        median_work_gap = (
            np.nan
        )

        q95_work_gap = (
            np.nan
        )

        avg_distance = (
            np.nan
        )

        median_distance = (
            np.nan
        )

        zero_distance_rate = (
            np.nan
        )

        avg_empty_time = (
            np.nan
        )

        median_empty_time = (
            np.nan
        )

        avg_slack = (
            np.nan
        )

        median_slack = (
            np.nan
        )

        q25_slack = (
            np.nan
        )

        q75_slack = (
            np.nan
        )


    # ========================================================
    # ORDER-LEVEL ACTUAL COMPARISON
    # ========================================================

    valid_compare = (
        recommended[
            recommended[
                "actual_empty_distance_km"
            ].notna()
            &
            recommended[
                "recommended_empty_distance_km"
            ].notna()
        ]
        .copy()
    )


    compare_n = (
        len(
            valid_compare
        )
    )


    if compare_n > 0:

        actual_total = (
            valid_compare[
                "actual_empty_distance_km"
            ].sum()
        )

        proposed_total = (
            valid_compare[
                "recommended_empty_distance_km"
            ].sum()
        )


        if actual_total > 0:

            saving_rate = (
                actual_total
                -
                proposed_total
            ) / actual_total

        else:

            saving_rate = (
                np.nan
            )


        improved_order_rate = (
            valid_compare[
                "recommended_empty_distance_km"
            ]
            <
            valid_compare[
                "actual_empty_distance_km"
            ]
        ).mean()


        mean_saving_distance = (
            valid_compare[
                "actual_empty_distance_km"
            ]
            -
            valid_compare[
                "recommended_empty_distance_km"
            ]
        ).mean()


        positive_actual = (
            valid_compare[
                valid_compare[
                    "actual_empty_distance_km"
                ]
                > 0
            ]
            .copy()
        )


        if len(
            positive_actual
        ) > 0:

            order_rates = (

                (
                    positive_actual[
                        "actual_empty_distance_km"
                    ]
                    -
                    positive_actual[
                        "recommended_empty_distance_km"
                    ]
                )
                /
                positive_actual[
                    "actual_empty_distance_km"
                ]

            )


            avg_order_saving_rate = (
                order_rates.mean()
            )

            median_order_saving_rate = (
                order_rates.median()
            )


        else:

            avg_order_saving_rate = (
                np.nan
            )

            median_order_saving_rate = (
                np.nan
            )


    else:

        actual_total = (
            np.nan
        )

        proposed_total = (
            np.nan
        )

        saving_rate = (
            np.nan
        )

        improved_order_rate = (
            np.nan
        )

        mean_saving_distance = (
            np.nan
        )

        avg_order_saving_rate = (
            np.nan
        )

        median_order_saving_rate = (
            np.nan
        )


    # ========================================================
    # VEHICLE-LEVEL METRICS
    # ========================================================

    vehicle_level = (
        make_vehicle_saving_table(
            result_df
        )
    )


    if len(
